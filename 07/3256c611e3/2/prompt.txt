```typescript
function performSearch() {
    // --- Clear previous highlights ---
    for (const pageData of pagesData) {
        pageData.textDivs.forEach((div, i) => restoreDiv(div, pageData.textContentItemsStr[i]));
    }
    highlightElements = [];
    matchCount = 0;
    currentMatchIndex = 0;

    if (!searchQuery.trim() || pagesData.length === 0) return;

    const query = matchCase ? searchQuery : searchQuery.toLowerCase();

    pagesData.forEach((pageData) => {
        const { textDivs, textContentItemsStr } = pageData;

        // Build charMap: charMap[absoluteIdx] = {divIdx, offset}
        interface CharPos { divIdx: number; offset: number; }
        const charMap: CharPos[] = [];
        let combinedText = '';
        for (let d = 0; d < textContentItemsStr.length; d++) {
            const s = textContentItemsStr[d];
            for (let o = 0; o < s.length; o++) charMap.push({ divIdx: d, offset: o });
            combinedText += s;
        }

        const textToSearch = matchCase ? combinedText : combinedText.toLowerCase();
        let searchPos = 0;

        // Collect converted matches: {begin, end (exclusive), matchIdx}
        interface ConvertedMatch { begin: CharPos; end: CharPos; matchIdx: number; }
        const convertedMatches: ConvertedMatch[] = [];

        while (true) {
            const matchStart = textToSearch.indexOf(query, searchPos);
            if (matchStart === -1) break;

            if (matchWholeWord) {
                const before = matchStart > 0 ? textToSearch[matchStart - 1] : ' ';
                const after = matchStart + query.length < textToSearch.length
                    ? textToSearch[matchStart + query.length] : ' ';
                if (/\w/.test(before) || /\w/.test(after)) { searchPos = matchStart + 1; continue; }
            }

            const endIdx = matchStart + query.length - 1;
            if (charMap[matchStart] && charMap[endIdx]) {
                convertedMatches.push({
                    begin: charMap[matchStart],
                    end: { divIdx: charMap[endIdx].divIdx, offset: charMap[endIdx].offset + 1 },
                    matchIdx: matchCount++
                });
            }
            searchPos = matchStart + 1;
        }

        if (convertedMatches.length === 0) return;

        // Render matches into DOM (mirrors TextHighlighter._renderMatches)
        let prevEnd: CharPos | null = null;

        for (const { begin, end, matchIdx } of convertedMatches) {
            // Flush tail of previous div if switching to a new one
            if (!prevEnd || begin.divIdx !== prevEnd.divIdx) {
                if (prevEnd !== null) {
                    appendTextToDiv(textDivs[prevEnd.divIdx], textContentItemsStr[prevEnd.divIdx],
                        prevEnd.offset, textContentItemsStr[prevEnd.divIdx].length);
                }
                beginDiv(textDivs[begin.divIdx], textContentItemsStr[begin.divIdx], begin.offset);
            } else {
                // Same div — fill gap between previous match end and this match start
                appendTextToDiv(textDivs[prevEnd.divIdx], textContentItemsStr[prevEnd.divIdx],
                    prevEnd.offset, begin.offset);
            }

            if (begin.divIdx === end.divIdx) {
                // Single-div match: inject inline highlight span
                const span = document.createElement('span');
                span.className = 'highlight appended';
                span.dataset.matchIndex = String(matchIdx);
                span.textContent = textContentItemsStr[begin.divIdx].substring(begin.offset, end.offset);
                textDivs[begin.divIdx].append(span);
                highlightElements.push(span);
            } else {
                // Multi-div match
                const beginSpan = document.createElement('span');
                beginSpan.className = 'highlight begin appended';
                beginSpan.dataset.matchIndex = String(matchIdx);
                beginSpan.textContent = textContentItemsStr[begin.divIdx].substring(begin.offset);
                textDivs[begin.divIdx].append(beginSpan);
                highlightElements.push(beginSpan);

                for (let n = begin.divIdx + 1; n < end.divIdx; n++) {
                    textDivs[n].className = 'highlight middle';
                    (textDivs[n] as HTMLElement).dataset.matchIndex = String(matchIdx);
                    highlightElements.push(textDivs[n]);
                }

                beginDiv(textDivs[end.divIdx], textContentItemsStr[end.divIdx], 0);
                const endSpan = document.createElement('span');
                endSpan.className = 'highlight end appended';
                endSpan.dataset.matchIndex = String(matchIdx);
                endSpan.textContent = textContentItemsStr[end.divIdx].substring(0, end.offset);
                textDivs[end.divIdx].append(endSpan);
                highlightElements.push(endSpan);
            }

            prevEnd = end;
        }

        // Flush tail of last div
        if (prevEnd !== null) {
            appendTextToDiv(textDivs[prevEnd.divIdx], textContentItemsStr[prevEnd.divIdx],
                prevEnd.offset, textContentItemsStr[prevEnd.divIdx].length);
        }
    });

    if (highlightElements.length > 0) scrollToMatch(0);
}
```

## 2. Key Technical Concepts

- **PDF.js TextLayer**: Renders invisible text spans on top of canvas for selection/search. Spans use `position: absolute`, `transform: scaleX(ratio)`, `font-size: calc(var(--total-scale-factor) * Npx)`, and `left/top` as percentages or `calc()` expressions.
- **`textLayer.textDivs`**: Array of `HTMLElement[]` — one span per text item after `await textLayer.render()`
- **`textLayer.textContentItemsStr`**: Array of strings 1:1 with textDivs
- **`pdf_viewer.css` rules relevant to highlights:**
  - Line 905: `.textLayer :is(span,br) { position: absolute }` — all descendant spans are absolutely positioned
  - Line 948–956: `.textLayer .highlight { margin: -1px; padding: 1px; border-radius: 4px; background-color: var(--highlight-bg-color) }`
  - Line 958: `.appended:is(.textLayer .highlight) { position: initial }` — overrides position to static for injected spans with `.appended` class
- **CSS transforms in PDF.js text spans:** `transform: scaleX(canvasWidth * scale / measuredWidth)` — applied after canvas measurement; `scale(1/minFontSize)` for browser min-font compensation
- **`Range.getBoundingClientRect()`** — unreliable for PDF.js spans because the combination of CSS variables, percentage positioning, and complex transforms causes incorrect character-level bounding box computation
- **`Element.getBoundingClientRect()`** — reliable for span-level visual bounding box, accounts for all CSS transforms
- **Kerning disruption** — splitting a text node into multiple DOM nodes changes kerning at boundaries; large fonts make this more visible
- **CharMap approach**: Building a combined text string and a `charMap` array mapping absolute character index → `{divIdx, offset}` for multi-item search matching
